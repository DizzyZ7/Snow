cmake_minimum_required(VERSION 3.24)
project(Snow VERSION 0.3.0 LANGUAGES CXX)
include(CTest)
option(SNOW_SANITIZERS "Enable supported address/undefined sanitizers" OFF)
option(SNOW_ANALYZE "Enable MSVC native static analysis" OFF)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  if(SNOW_SANITIZERS)
    # VS 17.7+ uses a dynamic ASan runtime even with the static CRT. CTest
    # launches outside MSBuild's compiler environment, so stage the matching DLL.
    get_filename_component(SNOW_COMPILER_DIRECTORY "${CMAKE_CXX_COMPILER}" DIRECTORY)
    find_file(SNOW_ASAN_RUNTIME clang_rt.asan_dynamic-x86_64.dll
      PATHS "${SNOW_COMPILER_DIRECTORY}" NO_DEFAULT_PATH REQUIRED)
    foreach(config Debug Release RelWithDebInfo MinSizeRel)
      file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/${config}")
      configure_file("${SNOW_ASAN_RUNTIME}" "${CMAKE_BINARY_DIR}/${config}/clang_rt.asan_dynamic-x86_64.dll" COPYONLY)
    endforeach()
  endif()
endif()
function(snow_target target)
  target_compile_features(${target} PUBLIC cxx_std_20)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /permissive- /utf-8)
    if(SNOW_ANALYZE)
      target_compile_options(${target} PRIVATE /analyze)
    endif()
    if(SNOW_SANITIZERS)
      target_compile_options(${target} PRIVATE /fsanitize=address /Zi)
      target_link_options(${target} PRIVATE /INCREMENTAL:NO)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(SNOW_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endfunction()
add_library(snow_core STATIC src/core.cpp)
target_include_directories(snow_core PUBLIC src)
snow_target(snow_core)
if(BUILD_TESTING)
  add_executable(snow_core_tests tests/core_tests.cpp)
  target_link_libraries(snow_core_tests PRIVATE snow_core)
  snow_target(snow_core_tests)
  add_test(NAME core COMMAND snow_core_tests)
endif()
if(WIN32)
  enable_language(RC)
  add_library(snow_windows STATIC src/renderer.cpp src/window_tracker.cpp)
  target_include_directories(snow_windows PUBLIC src)
  target_compile_definitions(snow_windows PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00 WINVER=0x0A00)
  target_link_libraries(snow_windows PUBLIC snow_core user32 gdi32 shell32 ole32 uuid dwmapi d3d11 dxgi dcomp d3dcompiler)
  snow_target(snow_windows)
  if(MSVC)
    # GDI startup must not eagerly load the optional shader compiler/GPU DLLs.
    target_link_libraries(snow_windows PUBLIC delayimp)
    target_link_options(snow_windows INTERFACE /DELAYLOAD:d3d11.dll /DELAYLOAD:dcomp.dll /DELAYLOAD:d3dcompiler_47.dll)
  endif()
  add_executable(Snow WIN32 src/main.cpp src/application.cpp src/Snow.rc)
  target_link_libraries(Snow PRIVATE snow_windows comctl32 wtsapi32 advapi32)
  snow_target(Snow)
  if(MSVC)
    target_link_options(Snow PRIVATE /MANIFEST:EMBED "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/src/Snow.manifest")
  endif()
  if(BUILD_TESTING)
    add_executable(snow_renderer_smoke tests/renderer_smoke.cpp)
    target_link_libraries(snow_renderer_smoke PRIVATE snow_windows)
    snow_target(snow_renderer_smoke)
    add_test(NAME renderer_gdi COMMAND snow_renderer_smoke)
    set_tests_properties(renderer_gdi PROPERTIES TIMEOUT 30)
    add_test(NAME renderer_alpha COMMAND snow_renderer_smoke --visual)
    set_tests_properties(renderer_alpha PROPERTIES TIMEOUT 30)
    add_test(NAME renderer_d3d11 COMMAND snow_renderer_smoke --d3d)
    set_tests_properties(renderer_d3d11 PROPERTIES TIMEOUT 30 SKIP_RETURN_CODE 77)
    add_test(NAME renderer_d3d11_alpha COMMAND snow_renderer_smoke --d3d --visual)
    set_tests_properties(renderer_d3d11_alpha PROPERTIES TIMEOUT 35 SKIP_RETURN_CODE 77)
    add_test(NAME profiler_counters COMMAND powershell -NoProfile -ExecutionPolicy Bypass
      -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/profiler_smoke.ps1")
    set_tests_properties(profiler_counters PROPERTIES TIMEOUT 25)
    add_executable(snow_input_smoke tests/input_smoke.cpp)
    target_link_libraries(snow_input_smoke PRIVATE snow_windows)
    snow_target(snow_input_smoke)
    add_test(NAME input_gdi COMMAND snow_input_smoke)
    add_test(NAME input_d3d11 COMMAND snow_input_smoke --d3d)
    set_tests_properties(input_gdi input_d3d11 PROPERTIES TIMEOUT 35)
    set_tests_properties(input_d3d11 PROPERTIES SKIP_RETURN_CODE 77)
    add_executable(snow_tracker_smoke tests/window_tracker_smoke.cpp)
    target_link_libraries(snow_tracker_smoke PRIVATE snow_windows)
    snow_target(snow_tracker_smoke)
    add_test(NAME window_tracker COMMAND snow_tracker_smoke)
    set_tests_properties(window_tracker PROPERTIES TIMEOUT 40)
    add_test(NAME app_lifecycle COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/tests/app_smoke.ps1" -Exe "$<TARGET_FILE:Snow>")
    set_tests_properties(app_lifecycle PROPERTIES TIMEOUT 40)
    set_tests_properties(renderer_gdi renderer_alpha renderer_d3d11 renderer_d3d11_alpha input_gdi input_d3d11 window_tracker app_lifecycle
      PROPERTIES RESOURCE_LOCK interactive_desktop)
  endif()
  install(TARGETS Snow RUNTIME DESTINATION .)
endif()
