cmake_minimum_required(VERSION 3.24)
project(Snow VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Snow currently supports Windows only")
endif()
add_executable(Snow WIN32 src/main.cpp)
target_compile_features(Snow PRIVATE cxx_std_20)
target_compile_definitions(Snow PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(Snow PRIVATE user32 gdi32 shell32)
if(MSVC)
  target_compile_options(Snow PRIVATE /W4 /permissive-)
endif()
